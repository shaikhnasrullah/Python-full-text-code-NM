14) Aim: To study polymorphism


Function Overloading


#include <iostream>
Using namespace std;


Class A {
Public:
    Void add(int x, int y) {
        Cout << x + y << endl;
    }


    Void add(double x, double y) {
        Cout << x + y << endl;
    }
};


Int main() {
    A a;
    a.add(1, 4);
    a.add(1.2, 3.8);


    return 0;
}






Operator Overloading


#include <iostream>
Using namespace std;


Class Demo {
Public:
    Int a;


    Demo operator+(Demo b) {
        Demo c;
        c.a = a + b.a;
        return c;
    }
};


Int main() {
    Demo x, y, z;


    x.a = 5;
    y.a = 10;


    z = x + y;


    cout << z.a;


    return 0;
}

15) Aim: To study inheritance


Single Inheritance


#include <iostream>
Using namespace std;


Class Person {
Public:
    Void showPerson() {
        Cout << “Person” << endl;
    }
};


Class Student : public Person {
Public:
    Void showStudent() {
        Cout << “Student” << endl;
    }
};


Int main() {
    Student s;
    s.showPerson();
    s.showStudent();


    return 0;
}






Multilevel Inheritance


#include <iostream>
Using namespace std;


Class Person {
Public:
    Void showPerson() {
        Cout << “Person” << endl;
    }
};


Class Student : public Person {
Public:
    Void showStudent() {
        Cout << “Student” << endl;
    }
};


Class Library : public Student {
Public:
    Void showLibrary() {
        Cout << “Library” << endl;
    }
};


Int main() {
    Library l;


    l.showPerson();
    l.showStudent();
    l.showLibrary();


    return 0;
}
